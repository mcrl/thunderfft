cmake_minimum_required(VERSION 3.18)
project(thunderfft_figure5 LANGUAGES CXX CUDA)
# Standalone Figure 5 build: cuFFT and ThunderFFT always; cuFFTDx with MATHDX_ROOT; cuDNN when found.
find_package(CUDAToolkit REQUIRED)
set(MATHDX_ROOT "" CACHE PATH "MathDx root containing include/cufftdx.hpp")
set(CUDNN_ROOT "" CACHE PATH "Optional cuDNN root containing include/cudnn.h")

list(LENGTH CMAKE_CUDA_ARCHITECTURES arch_count)
if(NOT arch_count EQUAL 1)
  message(FATAL_ERROR "Figure 5 needs exactly one CUDA architecture; got '${CMAKE_CUDA_ARCHITECTURES}'")
endif()
string(REGEX MATCH "^[0-9]+" fig5_arch "${CMAKE_CUDA_ARCHITECTURES}")

add_executable(thff_conv2d conv2d/benchmark.cu conv2d/cufft_conv.cu conv2d/thunderfft_c2c_conv.cu)
set_target_properties(thff_conv2d PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17)
target_include_directories(thff_conv2d PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/conv2d ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_link_libraries(thff_conv2d PRIVATE CUDA::cudart CUDA::cufft)
target_compile_options(thff_conv2d PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo> $<$<COMPILE_LANGUAGE:CXX>:-O3>)

if(EXISTS "${MATHDX_ROOT}/include/cufftdx.hpp")
  target_sources(thff_conv2d PRIVATE conv2d/cufftdx_c2c_conv.cu)
  target_include_directories(thff_conv2d PRIVATE ${MATHDX_ROOT}/include ${MATHDX_ROOT}/external/cutlass/include)
  target_compile_definitions(thff_conv2d PRIVATE THFF_WITH_CUFFTDX=1 THFF_CUFFTDX_SM=${fig5_arch}0)
  set(THFF_CONV2D_CUFFTDX ON CACHE INTERNAL "Figure 5 cuFFTDx backend" FORCE)
else()
  set(THFF_CONV2D_CUFFTDX OFF CACHE INTERNAL "Figure 5 cuFFTDx backend" FORCE)
endif()

find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS ${CUDNN_ROOT} PATH_SUFFIXES include)
find_library(CUDNN_LIBRARY cudnn HINTS ${CUDNN_ROOT} PATH_SUFFIXES lib lib64)
if(CUDNN_INCLUDE_DIR AND CUDNN_LIBRARY)
  target_sources(thff_conv2d PRIVATE conv2d/cudnn_conv.cu)
  target_include_directories(thff_conv2d BEFORE PRIVATE ${CUDNN_INCLUDE_DIR})
  target_link_libraries(thff_conv2d PRIVATE ${CUDNN_LIBRARY})
  target_compile_definitions(thff_conv2d PRIVATE THFF_WITH_CUDNN=1)
  set_target_properties(thff_conv2d PROPERTIES BUILD_RPATH "${CUDAToolkit_LIBRARY_DIR}")
  set(THFF_CONV2D_CUDNN ON CACHE INTERNAL "Figure 5 cuDNN backend" FORCE)
else()
  set(THFF_CONV2D_CUDNN OFF CACHE INTERNAL "Figure 5 cuDNN backend" FORCE)
endif()
